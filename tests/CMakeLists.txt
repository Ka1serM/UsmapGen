find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(USMAPGEN_TEST_EXECUTABLES ${CMAKE_CURRENT_SOURCE_DIR}/Executables CACHE PATH "The download folder of the dedicated server executables")
set(fetch ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/fetch.py ${USMAPGEN_TEST_EXECUTABLES})
add_test(NAME fetch/DepotDownloader COMMAND ${fetch})
set_tests_properties(fetch/DepotDownloader PROPERTIES FIXTURES_SETUP DepotDownloader TIMEOUT 600)

set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Executables.json)
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/Executables.json executables)
string(JSON count LENGTH ${executables})
math(EXPR last "${count} - 1")
foreach(index RANGE ${last})
    string(JSON engine GET ${executables} ${index} engine)
    string(JSON app GET ${executables} ${index} app)
    string(JSON file GET ${executables} ${index} file)
    get_filename_component(file_name ${file} NAME)
    get_filename_component(stem ${file} NAME_WE)
    set(name ${engine}/${stem})
    add_test(NAME fetch/${name} COMMAND ${fetch} ${app})
    set_tests_properties(fetch/${name} PROPERTIES FIXTURES_SETUP ${name} FIXTURES_REQUIRED DepotDownloader TIMEOUT 1800 LABELS fetch)
    add_test(NAME mappings/${name}
        COMMAND usmapgen-executable-test ${USMAPGEN_TEST_EXECUTABLES}/${app}/${file_name})
    set_tests_properties(mappings/${name} PROPERTIES FIXTURES_REQUIRED ${name} TIMEOUT 300 LABELS mappings)
endforeach()
